019def digest(path):
020    return hashlib.sha256(path.read_bytes()).hexdigest()

022def run(work, flow_path):
023    spec = importlib.util.spec_from_file_location('flowlens_samplescope', flow_path)
024    flow = importlib.util.module_from_spec(spec)
025    sys.modules[spec.name] = flow
026    spec.loader.exec_module(flow)
027    results = []
028    for key, label, byte_range, expected in RANGES:
029        raw = work / f'ctu13-botnet42-{key}.bin'
030        source = work / f'ctu13-botnet42-{key}-complete.binetflow'
031        if digest(raw) != expected:
032            raise ValueError(f'{key}: preserved source fingerprint changed')
033        raw_lines = raw.read_bytes().splitlines()
034        input_rows = source.read_bytes().splitlines()[1:]
035        derived_rows_match = any(input_rows == raw_lines[start:end]
036                                 for start in (0, 1) for end in (len(raw_lines), len(raw_lines) - 1))
037        if not derived_rows_match:
038            raise ValueError(f'{key}: input rows do not match the preserved byte range')
039        with source.open(encoding='utf-8-sig', newline='') as handle:
040            rows = list(csv.DictReader(handle))
041        protocols = Counter(row['Proto'].strip().upper() for row in rows)
042        botnet = sum(row['Label'].strip().lower().removeprefix('flow=').startswith('from-botnet') for row in rows)
043        byte_total = sum(int(row['TotBytes']) for row in rows)
044        parsed, invalid, _ = flow.load_flows(source)
045        eligible = []
046        excluded = Counter()
047        for row in rows:
048            try:
049                ipaddress.ip_address(row['SrcAddr'].strip())
050                ipaddress.ip_address(row['DstAddr'].strip())
051            except ValueError:
052                excluded[row['Proto'].upper()] += 1
053                continue
054            eligible.append(row)
055        crosscheck = (len(parsed) == len(eligible) and len(invalid) == sum(excluded.values())
056                      and Counter(r.protocol.upper() for r in parsed) == Counter(r['Proto'].upper() for r in eligible)
057                      and sum(r.byte_count for r in parsed) == sum(int(r['TotBytes']) for r in eligible)
058                      and sum(r.label_family == 'From-Botnet' for r in parsed) == botnet)
059        if not crosscheck:
060            raise ValueError(f'{key}: CSV aggregate and FlowLens parser disagree')
061        results.append(dict(segment=key, label=label, byte_range=byte_range, records=len(rows),
062                            flowlens_excluded=len(invalid), excluded_protocols=dict(excluded), flowlens_records=len(parsed), protocols=dict(sorted(protocols.items())),
063                            from_botnet=botnet, from_botnet_pct=round(botnet / len(rows) * 100, 4),
064                            total_bytes=byte_total, input_sha256=digest(source), raw_sha256=expected,
065                            derived_rows_match_raw=derived_rows_match, parser_crosscheck=crosscheck))
066    return dict(source_url=SOURCE, method='Preserved non-random byte slices; direct CSV versus FlowLens parsing',
067                limitation='Three selected byte ranges are not random samples and do not estimate population attack prevalence.',
068                total_records=sum(r['records'] for r in results), segments=results)
