037def parse_timestamp(value: str, ctu_format: bool = False) -> datetime:
038    cleaned = value.strip()
039    try:
040        result = datetime.fromisoformat(cleaned.replace("Z", "+00:00"))
041        if result.tzinfo is None or result.utcoffset() is None:
042            raise ValueError("timestamp must include a timezone")
043        return result
044    except ValueError:
045        if ctu_format:
046            for pattern in ("%Y/%m/%d %H:%M:%S.%f", "%Y/%m/%d %H:%M:%S"):
047                try:
048                    # CTU-13 StartTime has no timezone field; retain the source clock without guessing.
049                    return datetime.strptime(cleaned, pattern)
050                except ValueError:
051                    pass
052        raise ValueError("timestamp must be timezone-aware ISO 8601 or a CTU-13 source timestamp") from None

072def parse_port(value: str, field: str) -> int | None:
073    cleaned = value.strip().lower()
074    if cleaned in {"", "?", "-", "na", "none"}:
075        return None
076    try:
077        number = int(cleaned, 16) if cleaned.startswith("0x") else int(cleaned)

080    if not 0 <= number <= 65535:

082    return number

092def label_family(value: str) -> str | None:
093    cleaned = value.strip()
094    if not cleaned:
095        return None
096    if not LABEL_RE.fullmatch(cleaned):
097        raise ValueError("Label contains unsupported characters or is too long")
098    normalized = cleaned.lower().removeprefix("flow=")
099    if normalized.startswith("from-botnet"):
100        return "From-Botnet"
101    if normalized.startswith("to-botnet"):
102        return "To-Botnet"
103    if normalized.startswith("from-normal"):
104        return "From-Normal"
105    if normalized.startswith("to-normal"):
106        return "To-Normal"
107    if "background" in normalized:
108        return "Background"
109    if "botnet" in normalized:
110        return "Botnet (other label)"
111    return "Other"

114def load_flows(path: Path) -> tuple[list[Flow], list[int], str]:
115    flows: list[Flow] = []
116    invalid_lines: list[int] = []
117    with path.open("r", encoding="utf-8-sig", newline="") as handle:
118        reader = csv.DictReader(handle)
119        headers = set(reader.fieldnames or [])
120        if set(FIELDS).issubset(headers):
121            schema = "standard CSV"
122            is_ctu = False
123        elif CTU_FIELDS.issubset(headers):
124            schema = "CTU-13/Argus flow CSV"
125            is_ctu = True
126        else:
127            expected = ", ".join(FIELDS)
128            ctu_expected = ", ".join(sorted(CTU_FIELDS))
129            raise ValueError(f"CSV must use the standard columns ({expected}) or CTU-13 columns ({ctu_expected})")
130        for row in reader:
131            try:
132                if is_ctu:
133                    flows.append(Flow(
134                        timestamp=parse_timestamp(row.get("StartTime") or "", ctu_format=True),
135                        src_ip=parse_ip(row.get("SrcAddr") or "", "SrcAddr"),
136                        dst_ip=parse_ip(row.get("DstAddr") or "", "DstAddr"),
137                        src_port=parse_port(row.get("Sport") or "", "Sport"),
138                        dst_port=parse_port(row.get("Dport") or "", "Dport"),
139                        protocol=parse_token(row.get("Proto") or "", "Proto"),
140                        byte_count=parse_uint(row.get("TotBytes") or "", "TotBytes"),
141                        action=None,
142                        label_family=label_family(row.get("Label") or ""),
143                    ))
144                else:
145                    action_value = row.get("action") or ""
146                    action = parse_token(action_value, "action") if action_value.strip() else None
147                    raw_label = row.get("label") or row.get("Label") or ""
148                    flows.append(Flow(
149                        timestamp=parse_timestamp(row.get("timestamp") or ""),
150                        src_ip=parse_ip(row.get("src_ip") or "", "src_ip"),
151                        dst_ip=parse_ip(row.get("dst_ip") or "", "dst_ip"),
152                        src_port=parse_port(row.get("src_port") or "", "src_port"),
153                        dst_port=parse_port(row.get("dst_port") or "", "dst_port"),
154                        protocol=parse_token(row.get("protocol") or "", "protocol"),
155                        byte_count=parse_uint(row.get("bytes") or "", "bytes"),
156                        action=action,
157                        label_family=label_family(raw_label) if raw_label.strip() else None,
158                    ))
159            except ValueError:
160                invalid_lines.append(reader.line_num)
161    return flows, invalid_lines, schema

164def network_aliases(flows: list[Flow]) -> dict[str, str]:
165    totals: dict[str, tuple[int, int]] = {}
166    for flow in flows:
167        for value in (flow.src_ip, flow.dst_ip):
168            address = ipaddress.ip_address(value)
169            prefix = 24 if address.version == 4 else 64
170            network = str(ipaddress.ip_network(f"{value}/{prefix}", strict=False))
171            count, byte_count = totals.get(network, (0, 0))
172            totals[network] = (count + 1, byte_count + flow.byte_count)
173    ordered = sorted(totals.items(), key=lambda item: (-item[1][1], -item[1][0], item[0]))
174    return {network: f"net-{number:02d}" for number, (network, _) in enumerate(ordered, 1)}

192def window_number(timestamp: datetime, window_minutes: int) -> int:
193    if timestamp.tzinfo is not None and timestamp.utcoffset() is not None:
194        seconds = timestamp.astimezone(timezone.utc).timestamp()

200def format_window_start(window: int, window_minutes: int, source_clock: bool) -> str:
201    seconds = window * window_minutes * 60
202    if source_clock:
203        value = datetime(1970, 1, 1) + timedelta(seconds=seconds)
204        return value.strftime("%Y-%m-%d %H:%M (source clock)")
205    value = datetime.fromtimestamp(seconds, tz=timezone.utc)
206    return value.strftime("%Y-%m-%d %H:%M UTC")

209def input_fingerprint(path: Path) -> tuple[int, str]:
210    digest = hashlib.sha256()
211    size = 0
212    with path.open("rb") as handle:
213        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
214            digest.update(chunk)
215            size += len(chunk)
216    return size, digest.hexdigest()
