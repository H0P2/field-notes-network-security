021def sha256(path: Path) -> str:
022    digest = hashlib.sha256()
023    with path.open("rb") as handle:
024        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
025            digest.update(chunk)
026    return digest.hexdigest()

029def parse_port(value: str) -> int | None:
030    cleaned = value.strip().lower()
031    if cleaned in MISSING_PORTS:
032        return None
033    port = int(cleaned, 16) if cleaned.startswith("0x") else int(cleaned)
034    if not 0 <= port <= 65535:
035        raise ValueError("port out of range")
036    return port

180def check_manifest(projects: Path, manifest_path: Path) -> tuple[list[str], bool]:
181    manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
182    results = []
183    for item in manifest["artifacts"]:
184        report_path = (manifest_path.parent / item["report_path"]).resolve()
