022def valid_public_url(value: object) -> bool:

036def validate_manifest(manifest_path: Path) -> tuple[list[str], list[str]]:
037    errors: list[str] = []
038    passed: list[str] = []
039    try:
040        manifest = json.loads(manifest_path.read_text(encoding="utf-8-sig"))
041    except (OSError, UnicodeError, json.JSONDecodeError):
042        return ["manifest를 읽을 수 없거나 유효한 UTF-8 JSON이 아닙니다"], passed
043    if not isinstance(manifest, dict) or manifest.get("format_version") != 1:
044        return ["manifest format_version 1이 필요합니다"], passed
045    artifacts = manifest.get("artifacts")
046    if not isinstance(artifacts, list) or not artifacts:
047        return ["artifacts 목록이 비어 있지 않아야 합니다"], passed

050    ids: set[str] = set()
051    for number, item in enumerate(artifacts, 1):
052        prefix = f"artifacts[{number}]"
053        if not isinstance(item, dict):
054            errors.append(f"{prefix}: 객체가 필요합니다")
055            continue
056        artifact_id = item.get("id")
057        if not nonempty(artifact_id) or artifact_id in ids:
058            errors.append(f"{prefix}.id: 중복되지 않는 이름이 필요합니다")
059            continue
060        ids.add(artifact_id)

081        expected_input_hash = item.get("input_sha256")
082        downloaded_hash = item.get("downloaded_range_sha256")
083        if not isinstance(expected_report_hash, str) or not SHA256_RE.fullmatch(expected_report_hash):
084            errors.append(f"{artifact_id}: report_sha256 형식이 올바르지 않습니다")
085        elif sha256_bytes(report_raw) != expected_report_hash:

089            errors.append(f"{artifact_id}: input_sha256 형식이 올바르지 않습니다")
090        elif f"입력 파일 SHA-256: `{expected_input_hash}`" not in report:
091            errors.append(f"{artifact_id}: 보고서에서 입력 파일 SHA-256을 확인할 수 없습니다")
092 
093        if not isinstance(downloaded_hash, str) or not SHA256_RE.fullmatch(downloaded_hash):
094            errors.append(f"{artifact_id}: downloaded_range_sha256 형식이 올바르지 않습니다")
095        elif f"내려받은 원본 구간 SHA-256: `{downloaded_hash}`" not in report:
096            errors.append(f"{artifact_id}: 보고서에서 내려받은 원본 구간 지문을 확인할 수 없습니다")
097        if not valid_public_url(item.get("source_url")):
098            errors.append(f"{artifact_id}: 출처는 자격 증명·query·fragment가 없는 공개 HTTPS URL이어야 합니다")
099        elif item["source_url"] not in report:

103        if not nonempty(sample_scope) or sample_scope not in report:

105        limitation = item.get("limitation")
106        if not nonempty(limitation) or f"표본 한계: {limitation}" not in report:
107            errors.append(f"{artifact_id}: 보고서와 manifest에 같은 해석 한계가 있어야 합니다")
108        if type(item.get("records_analyzed")) is not int or item["records_analyzed"] < 1:

114            if not count_line or int(count_line.group(1).replace(",", "")) != item["records_analyzed"]:
115                errors.append(f"{artifact_id}: records_analyzed와 보고서 기록 수가 다릅니다")
116        if type(item.get("input_bytes")) is not int or item["input_bytes"] < 1:
117            errors.append(f"{artifact_id}: input_bytes는 양의 정수여야 합니다")
118        elif f"입력 파일 크기: {item['input_bytes']:,} bytes" not in report:
119            errors.append(f"{artifact_id}: 보고서 입력 크기와 manifest가 다릅니다")
120        if item.get("identifiers_redacted") is not True:
121            errors.append(f"{artifact_id}: 식별자 비식별화 확인이 필요합니다")
122        if IPV4_RE.search(report) or EMAIL_RE.search(report):
123            errors.append(f"{artifact_id}: 보고서에 원문 IPv4 또는 이메일처럼 보이는 값이 있습니다")
124 
125        artifact_errors = [error for error in errors if error.startswith(f"{artifact_id}:")]
126        if not artifact_errors:

132def main(argv: list[str] | None = None) -> int:
133    parser = argparse.ArgumentParser(description="실제 분석 결과의 파일 무결성과 출처·범위·비식별화 정보를 확인합니다.")
134    parser.add_argument("manifest", type=Path, help="evidence manifest JSON")
135    parser.add_argument("-o", "--output", type=Path, help="확인 결과 Markdown 저장 경로 (선택)")
136    args = parser.parse_args(argv)
137    errors, passed = validate_manifest(args.manifest)
138    lines = ["# PublishGate - 실제 증거 패키지 확인", ""]
139    for item in passed:
140        lines.append(f"- PASS: {item}")
141    for item in errors:
142        lines.append(f"- FAIL: {item}")
143    if errors:
144        lines.extend(["", f"결과: FAIL ({len(errors)}개 확인 항목)"])
145        exit_code = 1
146    else:
147        lines.extend(["", f"결과: PASS ({len(passed)}개 보고서 확인)"])
148        exit_code = 0
149    report = "\n".join(lines) + "\n"
150    print(report, end="")
151    if args.output:
152        try:
153            args.output.parent.mkdir(parents=True, exist_ok=True)
154            args.output.write_text(report, encoding="utf-8")
155        except OSError as exc:
156            print(f"PublishGate 오류: 결과를 저장할 수 없습니다 ({exc})", file=sys.stderr)
157            return 2
158    return exit_code
