053def load_cowrie_jsonl(path: Path) -> tuple[list[SessionSummary], list[int]]:
054    sessions: list[SessionSummary] = []
055    invalid: list[int] = []
056    with open_text(path) as handle:
057        for line_number, line in enumerate(handle, 1):
058            if not line.strip():
059                continue
060            try:
061                record = json.loads(line)
062                if not isinstance(record, dict):
063                    raise ValueError("record must be an object")
064                source = record.get("src_ip")
065                if not isinstance(source, str) or not source.strip():
066                    raise ValueError("src_ip is required")
067                sessions.append(SessionSummary(
068                    first_seen_day=parse_first_seen(record.get("first_seen")),
069                    source=source.strip(),
070                    failed_logins=parse_count(record.get("cnt_login_failed", 0), "cnt_login_failed"),
071                    successful_logins=parse_count(record.get("cnt_login_success", 0), "cnt_login_success"),
072                ))
073            except (json.JSONDecodeError, ValueError):
074                invalid.append(line_number)
075    return sessions, invalid

163def input_fingerprint(path: Path) -> tuple[int, str]:
164    digest = hashlib.sha256()
165    size = 0
166    with path.open("rb") as handle:
167        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
168            digest.update(chunk)
169            size += len(chunk)
170    return size, digest.hexdigest()

173def main(argv: list[str] | None = None) -> int:
