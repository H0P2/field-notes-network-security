049def scan_file(path: Path, root: Path) -> tuple[list[Finding], str]:
050    try:
051        if path.stat().st_size > MAX_FILE_BYTES:
052            return [], "skipped_oversized"
053        raw = path.read_bytes()
054    except OSError:
055        return [], "unreadable_file"
056    if b"\0" in raw:
057        return [], "skipped_binary"
058    try:
059        content = raw.decode("utf-8-sig")
060    except UnicodeDecodeError:
061        return [], "skipped_non_utf8"
062    relative = path.relative_to(root).as_posix().replace("|", "\\|")
063    findings: list[Finding] = []
064    for line_number, line in enumerate(content.splitlines(), 1):
065        for category, pattern in PATTERNS:
066            if pattern.search(line):
067                findings.append(Finding(relative, line_number, category))
068    return findings, "scanned"

071def files_under(root: Path, counts: dict[str, int]) -> Iterator[Path]:
072    for path in root.rglob("*"):
073        try:
074            relative_parts = path.relative_to(root).parts
075        except ValueError:
076            continue
077        if any(part in IGNORED_DIRS for part in relative_parts):
078            continue
079        if path.is_symlink():
080            counts["skipped_symlink"] += 1
081            continue
082        if path.is_file():
083            yield path

086def scan_worktree(root: Path) -> tuple[list[Finding], dict[str, int]]:
087    findings: set[Finding] = set()
088    counts = {"scanned": 0, "skipped_oversized": 0, "skipped_binary": 0, "skipped_non_utf8": 0, "skipped_symlink": 0, "unreadable_file": 0}
089    for path in files_under(root, counts):
090        found, status = scan_file(path, root)
091        findings.update(found)
092        counts[status] += 1
093    return sorted(findings), counts

096def scan_history(root: Path, commit_limit: int) -> tuple[list[Finding], bool]:
097    try:
098        git = subprocess.run(
099            ["git", "-C", str(root), "rev-list", "--all", f"--max-count={commit_limit + 1}"],
100            capture_output=True, text=True, encoding="utf-8", errors="replace", check=False,
101        )
102    except OSError as exc:
103        if isinstance(exc, FileNotFoundError):
104            raise RuntimeError("Git executable is not available; history scan was not performed") from None
105        raise RuntimeError("Git could not be started; history scan was not performed") from None
106    if git.returncode != 0:
107        raise RuntimeError("Git history could not be read; check that the folder is a Git repository and Git is installed")
108    commits = git.stdout.splitlines()
109    truncated = len(commits) > commit_limit
110    commits = commits[:commit_limit]
111    findings: set[Finding] = set()
112    for commit in commits:
113        command = ["git", "-C", str(root), "grep", "--no-color", "-n", "-I", "-E"]
114        for pattern in GIT_PATTERNS:
115            command.extend(["-e", pattern])
116        command.extend([commit, "--"])
117        process = subprocess.Popen(command, stdout=subprocess.PIPE, stderr=subprocess.DEVNULL, text=True, encoding="utf-8", errors="replace")
118        assert process.stdout is not None
119        for output_line in process.stdout:
120            parts = output_line.rstrip("\r\n").split(":", 3)
121            if len(parts) < 3:
122                continue
123            try:
124                line_number = int(parts[2])
125            except ValueError:
126                continue
127            path = parts[1].replace("|", "\\|")
128            findings.add(Finding(f"git:{commit[:10]}/{path}", line_number, "history_match"))
129        return_code = process.wait()
130        # git grep returns 1 when the commit has no matches; other non-zero codes are errors.
131        if return_code not in (0, 1):
132            raise RuntimeError("Git could not inspect one or more commits")
133    return sorted(findings), truncated
